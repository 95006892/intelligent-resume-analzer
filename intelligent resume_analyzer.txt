/* ======================================================================
   INTELLIGENT RESUME ANALYZER — matching engine
   Everything below uses only built-in JavaScript (ES6+) features:
   RegExp, Map, Set, Math, Array/String prototype methods.
   No external libraries, packages, or network calls anywhere.

const SKILL_DB = {
  "Programming": ["python","java","javascript","typescript","c++","c#"," c ","sql","r ","go","golang","ruby","php","swift","kotlin","scala","matlab","perl","html","css"],
  "Web & Frameworks": ["react","angular","vue","django","flask","node.js","nodejs","express","spring","spring boot","laravel","bootstrap","jquery","next.js","fastapi","rest api"],
  "Databases": ["mysql","postgresql","mongodb","oracle","sqlite","redis","cassandra","dynamodb","firebase"],
  "Cloud & DevOps": ["aws","azure","gcp","docker","kubernetes","jenkins","git","github","gitlab","ci/cd","linux","terraform"],
  "Data & AI": ["machine learning","deep learning","tensorflow","pytorch","keras","pandas","numpy","scikit-learn","nlp","computer vision","data analysis","data science","tableau","power bi","excel","statistics"],
  "Soft Skills": ["leadership","teamwork","communication","problem solving","time management","adaptability","creativity","collaboration","critical thinking","project management"]
};

const EDU_TERMS = ["b.tech","btech","m.tech","mtech","b.e.","b.e","m.e.","m.e","bachelor","master","mba","phd","b.sc","bsc","m.sc","msc","mca","bca","diploma"];

const STOPWORDS = new Set(("a an the and or but if then else for to of in on at by with from as is are was were be been being "+
  "this that these those it its into your you we our their they he she his her i my me will would can could should shall "+
  "have has had do does did not no yes so such than too very just about over under above below up down out off again "+
  "further here there when where why how all any both each few more most other some own same s t don now years experience "+
  "work team looking candidate role job description responsibilities requirements skills required preferred strong ability "+
  "excellent knowledge including etc use used using per year per").split(" "));

function normalize(text){
  return (text || "").toLowerCase();
}

function tokenize(text){
  const norm = normalize(text).replace(/[^a-z0-9+#. ]/g," ");
  return norm.split(/\s+/).filter(w => w.length > 2 && !STOPWORDS.has(w));
}

function extractSkills(text){
  const norm = " " + normalize(text).replace(/[,;/()]/g," ") + " ";
  const found = {};
  for (const cat in SKILL_DB){
    found[cat] = [];
    SKILL_DB[cat].forEach(skill => {
      const s = skill.trim();
      if (s.length === 0) return;
      if (norm.includes(" " + s) || norm.includes(s + " ")) {
        found[cat].push(s.trim());
      }
    });
  }
  return found;
}

function flatten(skillObj){
  let out = [];
  for (const cat in skillObj) out = out.concat(skillObj[cat]);
  return out;
}

function wordFreq(tokens){
  const m = new Map();
  tokens.forEach(t => m.set(t, (m.get(t)||0)+1));
  return m;
}

function cosineSimilarity(freqA, freqB){
  const vocab = new Set([...freqA.keys(), ...freqB.keys()]);
  let dot=0, magA=0, magB=0;
  vocab.forEach(w=>{
    const a = freqA.get(w)||0, b = freqB.get(w)||0;
    dot += a*b; magA += a*a; magB += b*b;
  });
  if (magA===0 || magB===0) return 0;
  return dot / (Math.sqrt(magA) * Math.sqrt(magB));
}

function extractYears(text){
  const m = normalize(text).match(/(\d+)\+?\s*(?:years|yrs)/);
  return m ? parseInt(m[1],10) : null;
}

function extractEducation(text){
  const norm = " " + normalize(text) + " ";
  return EDU_TERMS.filter(t => norm.includes(" "+t) || norm.includes(t+" "));
}

function analyze(jdText, resumeText){
  const jdSkills = extractSkills(jdText);
  const resSkills = extractSkills(resumeText);

  const jdFlat = flatten(jdSkills);
  const resFlat = new Set(flatten(resSkills));

  const matched = jdFlat.filter(s => resFlat.has(s));
  const missing = jdFlat.filter(s => !resFlat.has(s));
  const skillScore = jdFlat.length ? (matched.length / jdFlat.length) * 100 : 50;

  const jdFreq = wordFreq(tokenize(jdText));
  const resFreq = wordFreq(tokenize(resumeText));
  const textSim = cosineSimilarity(jdFreq, resFreq) * 100;

  const jdYears = extractYears(jdText);
  const resYears = extractYears(resumeText);
  let expScore = 70; // neutral if JD doesn't specify
  if (jdYears !== null){
    if (resYears === null) expScore = 30;
    else expScore = Math.min(100, (resYears / jdYears) * 100);
  }

  const jdEdu = extractEducation(jdText);
  const resEdu = extractEducation(resumeText);
  const eduMatch = jdEdu.length === 0 ? true : jdEdu.some(e => resEdu.includes(e));

  const overall = Math.round(
    skillScore * 0.55 +
    textSim * 0.30 +
    expScore * 0.15
  );

  const catBreakdown = {};
  for (const cat in jdSkills){
    const total = jdSkills[cat].length;
    if (total === 0) continue;
    const got = jdSkills[cat].filter(s => resFlat.has(s)).length;
    catBreakdown[cat] = Math.round((got/total)*100);
  }

  return {
    overall: Math.max(0, Math.min(100, overall)),
    skillScore: Math.round(skillScore),
    textSim: Math.round(textSim),
    expScore: Math.round(expScore),
    matched, missing, catBreakdown,
    jdYears, resYears, eduMatch, jdEdu, resEdu
  };
}

function tier(score){
  if (score >= 75) return {cls:"tier-strong", label:"Strong Match"};
  if (score >= 50) return {cls:"tier-moderate", label:"Moderate Match"};
  return {cls:"tier-weak", label:"Weak Match"};
}
function tierColor(score){
  if (score >= 75) return "#1f7a52";
  if (score >= 50) return "#c99328";
  return "#c1543a";
}

/* ---------------- Rendering: single analysis ---------------- */
function renderResult(container, r){
  const t = tier(r.overall);
  let suggestion = "";
  if (r.missing.length){
    suggestion = "Consider highlighting or gaining experience in: " + r.missing.slice(0,6).join(", ") + (r.missing.length>6 ? ", …" : "") + ".";
  } else {
    suggestion = "This resume covers every skill keyword found in the job description.";
  }
  if (r.jdYears !== null){
    if (r.resYears === null) suggestion += " The JD asks for " + r.jdYears + "+ years — no clear experience duration was found in the resume.";
    else if (r.resYears < r.jdYears) suggestion += " The JD asks for " + r.jdYears + "+ years; resume shows about " + r.resYears + ".";
  }

  const barsHtml = Object.keys(r.catBreakdown).map(cat=>{
    const val = r.catBreakdown[cat];
    return `<div class="barrow">
      <div class="cat">${cat}</div>
      <div class="track"><div class="fill" data-w="${val}" style="background:${tierColor(val)}"></div></div>
      <div class="num">${val}%</div>
    </div>`;
  }).join("");

  container.innerHTML = `
    <div class="result-head">
      <div class="stamp ${t.cls}">
        <div class="pct">${r.overall}%</div>
        <div class="lbl">${t.label}</div>
      </div>
      <div class="verdict">
        <h2>${t.label}</h2>
        <p>${suggestion}</p>
      </div>
    </div>
    <div class="barrows">${barsHtml || '<p style="color:#8a92a3;font-size:13px;">No tracked skill keywords were detected in the job description.</p>'}</div>
    <div class="chipsec">
      <h3>Matched skills (${r.matched.length})</h3>
      <div class="chips">${r.matched.map(s=>`<span class="chip match">${s}</span>`).join("") || '<span style="color:#8a92a3;font-size:13px;">None found</span>'}</div>
      <h3>Missing skills (${r.missing.length})</h3>
      <div class="chips">${r.missing.map(s=>`<span class="chip miss">${s}</span>`).join("") || '<span style="color:#8a92a3;font-size:13px;">None — full coverage</span>'}</div>
    </div>
  `;
  requestAnimationFrame(()=>{
    container.querySelectorAll(".fill").forEach(f=>{ f.style.width = f.dataset.w + "%"; });
  });
}

  });
}
function wireFileLoad(fileId, taId){
  document.getElementById(fileId).addEventListener("change", e=>{
    const f = e.target.files[0];
    if (!f) return;

let slotCount = 0;

/* Bulk upload: pick many resume .txt files at once (or a whole folder), each becomes its own slot */
let bulkLoadedCount = 0;
function loadFilesAsSlots(fileList){
  const files = [...fileList].filter(f => f.name.toLowerCase().endsWith(".txt"));
  if (!files.length){
    alert("No .txt files found in your selection. Only plain-text (.txt) resumes are supported.");
    return;
  }
  files.forEach(f=>{
    const reader = new FileReader();
    reader.onload = ev=>{
      const baseName = f.name.split("/").pop().replace(/\.[^/.]+$/, "");
      addSlot(baseName, ev.target.result);
      bulkLoadedCount++;
      document.getElementById("bulk-count").textContent = bulkLoadedCount + " loaded";
    };
    reader.readAsText(f);
  });
}
document.getElementById("bulk-file").addEventListener("change", e=>{
  loadFilesAsSlots(e.target.files);
  e.target.value = ""; // allow re-selecting the same files later
});
document.getElementById("folder-file").addEventListener("change", e=>{
  loadFilesAsSlots(e.target.files);
  e.target.value = "";
});

document.getElementById("rank-btn").addEventListener("click", ()=>{
  const jd = document.getElementById("jd-text-b").value;
  if (!jd.trim()){ alert("Please provide a job description."); return; }
  const slots = [...document.querySelectorAll(".resume-slot")];
  const entries = slots.map(s=>({
    name: s.querySelector(".slot-name").value.trim() || "Unnamed candidate",
    text: s.querySelector(".slot-text").value
  })).filter(e => e.text.trim().length > 0);

  if (!entries.length){ alert("Add at least one resume with text."); return; }

  const results = entries.map(e => ({ name: e.name, r: analyze(jd, e.text) }))
                          .sort((a,b) => b.r.overall - a.r.overall);

  renderBatch(document.getElementById("batch-results"), jd, results);
});

function renderBatch(container, jd, results){
  const rows = results.map((item, idx)=>{
    const t = tier(item.r.overall);
    return `
    <tr class="rankrow" data-idx="${idx}">
      <td class="rank-num">#${idx+1}</td>
      <td>${item.name}</td>
      <td style="width:160px;">
        <div class="mini-track"><div class="mini-fill" style="width:${item.r.overall}%;background:${tierColor(item.r.overall)}"></div></div>
      </td>
      <td style="width:60px;font-family:var(--mono);">${item.r.overall}%</td>
      <td style="width:130px;"><span class="tier-pill" style="background:${tierColor(item.r.overall)}22;color:${tierColor(item.r.overall)};">${t.label}</span></td>
    </tr>
    <tr class="detail-row" id="detail-${idx}" style="display:none;"><td colspan="5"><div class="detail-inner"></div></td></tr>
    `;
  }).join("");

  container.innerHTML = `
    <h3 style="font-family:var(--mono);font-size:11.5px;letter-spacing:0.08em;text-transform:uppercase;color:var(--ink-soft);margin-bottom:12px;">
      Ranked candidates — click a row for details
    </h3>
    <table class="rank">
      <thead><tr><th>Rank</th><th>Candidate</th><th>Score</th><th></th><th>Verdict</th></tr></thead>
      <tbody>${rows}</tbody>
    </table>
  `;

  container.querySelectorAll(".rankrow").forEach(row=>{
    row.addEventListener("click", ()=>{
      const idx = row.dataset.idx;
      const detailRow = document.getElementById("detail-"+idx);
      const isOpen = detailRow.style.display !== "none";
      container.querySelectorAll(".detail-row").forEach(d=> d.style.display="none");
      if (!isOpen){
        detailRow.style.display = "";
        renderResult(detailRow.querySelector(".detail-inner"), results[idx].r);
      }
    });
  });
}
